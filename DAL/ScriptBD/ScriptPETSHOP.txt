-- 2. Para actualizar el DVV en la tabla de control vertical
CREATE PROCEDURE [dbo].[sp_ActualizarDVV]
    @NombreTabla VARCHAR(100),
    @DVV INT -- O el tipo de dato que maneje tu DVV
AS
BEGIN
    SET NOCOUNT ON;

    -- Si ya existe el registro de la tabla, lo actualiza; si no, lo inserta
    IF EXISTS (SELECT 1 FROM [dbo].[DV_VERTICAL] WHERE [NombreTabla] = @NombreTabla)
    BEGIN
        UPDATE [dbo].[DV_VERTICAL]
        SET [DigitoVertical] = @DVV
        WHERE [NombreTabla] = @NombreTabla;
    END
    ELSE
    BEGIN
        INSERT INTO [dbo].[DV_VERTICAL] ([NombreTabla], [DigitoVertical])
        VALUES (@NombreTabla, @DVV);
    END
END

GO

CREATE PROCEDURE [dbo].[sp_ListarBackups]
    @Db NVARCHAR(100)
AS
BEGIN
    SELECT 
        bmf.physical_device_name AS RutaArchivo,
        bs.backup_finish_date    AS Fecha,
        bs.database_name         AS NombreBaseDatos
    FROM msdb.dbo.backupset bs
    JOIN msdb.dbo.backupmediafamily bmf
        ON bs.media_set_id = bmf.media_set_id
    WHERE bs.database_name = @Db
    ORDER BY bs.backup_finish_date DESC;
END
